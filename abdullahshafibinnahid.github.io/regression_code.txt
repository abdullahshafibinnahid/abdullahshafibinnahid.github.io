It is the code of a kaggle competition : Housing Prices : Advanced Regression Techniques. It includes XGBRegrssor and feature engineering.

import pandas as pd
import numpy as np
from xgboost import XGBRegressor
train = pd.read_csv('/kaggle/input/competitions/house-prices-advanced-regression-techniques/train.csv')
test = pd.read_csv('/kaggle/input/competitions/house-prices-advanced-regression-techniques/test.csv')

X_train = train.drop(columns=['SalePrice', 'Id'])
y_train = np.log1p(train['SalePrice'])
X_test = test.drop(columns=['Id'])
X_train = pd.get_dummies(X_train)
X_test = pd.get_dummies(X_test)
X_train, X_test = X_train.align(
    X_test,
    join='left',
    axis=1,
    fill_value=0
)
X_train = X_train.fillna(0)
X_test = X_test.fillna(0)
model = XGBRegressor(
    n_estimators=5000,
    learning_rate=0.01,
    max_depth=3,
    subsample=0.8,
    colsample_bytree=0.8,
    reg_alpha=0.0005,
    reg_lambda=0.9,
    random_state=42
)
X_train['TotalSF'] = (
    X_train['TotalBsmtSF'] +
    X_train['1stFlrSF'] +
    X_train['2ndFlrSF']
)

X_test['TotalSF'] = (
    X_test['TotalBsmtSF'] +
    X_test['1stFlrSF'] +
    X_test['2ndFlrSF']
)

X_train['HouseAge'] = (
    X_train['YrSold'] - X_train['YearBuilt']
)

X_test['HouseAge'] = (
    X_test['YrSold'] - X_test['YearBuilt']
)
X_train['TotalBath'] = (
    X_train['FullBath'] +
    0.5 * X_train['HalfBath'] +
    X_train['BsmtFullBath'] +
    0.5 * X_train['BsmtHalfBath']
)

X_test['TotalBath'] = (
    X_test['FullBath'] +
    0.5 * X_test['HalfBath'] +
    X_test['BsmtFullBath'] +
    0.5 * X_test['BsmtHalfBath']
)
X_train['Remodeled'] = (
    X_train['YearRemodAdd'] != X_train['YearBuilt']
).astype(int)

X_test['Remodeled'] = (
    X_test['YearRemodAdd'] != X_test['YearBuilt']
).astype(int)
X_train['OverallQualSF'] = (
    X_train['OverallQual'] * X_train['GrLivArea']
)

X_test['OverallQualSF'] = (
    X_test['OverallQual'] * X_test['GrLivArea']
)

model.fit(X_train, y_train)
y_pred = model.predict(X_test)
y_test = np.expm1(y_pred)
submission = pd.DataFrame({
    'Id': test['Id'],
    'SalePrice': y_test
})
submission.to_csv('price.csv', index=False)
